/* Script to create roles for SLGS */
CREATE ROLE slgs_migrator
    LOGIN
    PASSWORD 'DataSam@2026.';

CREATE ROLE slgs_web
    LOGIN
    PASSWORD 'DataSam@2026.';

CREATE ROLE slgs_cms
    LOGIN
    PASSWORD 'DataSam@2026.';

CREATE ROLE slgs_sims
    LOGIN
    PASSWORD 'DataSam@2026.';

/* Grant connect to database */
GRANT CONNECT ON DATABASE slgs TO slgs_migrator, slgs_web, slgs_cms, slgs_sims;

/* Grant usage on schema */
GRANT USAGE ON SCHEMA public TO slgs_web, slgs_cms, slgs_sims;

/* Grant usage and create on schema to migrator */
GRANT USAGE, CREATE ON SCHEMA public
TO slgs_migrator;

/* Grant all privileges on all tables in schema to migrator */
GRANT ALL PRIVILEGES ON ALL TABLES IN SCHEMA public
TO slgs_migrator;

/* Grant all privileges on all sequences in schema to migrator */
GRANT ALL PRIVILEGES ON ALL SEQUENCES IN SCHEMA public
TO slgs_migrator;

/* Grant connect and create on database to migrator */
GRANT CONNECT ON DATABASE slgs TO slgs_migrator;
GRANT CREATE ON DATABASE slgs TO slgs_migrator;

/* Grant usage and create on schema to migrator */
GRANT USAGE, CREATE ON SCHEMA public TO slgs_migrator;

/* Create identity schema */
CREATE SCHEMA "identity";

/* Drop identity schema */
DROP SCHEMA "identity" CASCADE;

/* Create platform admin role */
CREATE ROLE slgs_platform_admin
    LOGIN
    PASSWORD 'DataSam@206.';

/* Grant connect to database */
GRANT CONNECT ON DATABASE slgs
TO slgs_platform_admin;

/* Grant usage on identity schema */
GRANT USAGE ON SCHEMA identity
TO slgs_platform_admin;

/* Grant connect to database */
GRANT CONNECT ON DATABASE slgs
TO slgs_platform_admin;

/* Grant usage on identity schema */
GRANT USAGE ON SCHEMA identity
TO slgs_platform_admin;

/* Grant all privileges on all tables in identity schema */
GRANT SELECT, INSERT, UPDATE, DELETE
ON ALL TABLES IN SCHEMA identity
TO slgs_platform_admin;

/* Grant usage on identity schema */
GRANT USAGE ON SCHEMA identity
TO slgs_platform_admin;

/* Grant all privileges on all tables in identity schema */
GRANT SELECT, INSERT, UPDATE, DELETE
ON ALL TABLES IN SCHEMA identity
TO slgs_platform_admin;
